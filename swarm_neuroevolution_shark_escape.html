import numpy as np
import matplotlib.pyplot as plt

# --- 1. ENVIRONMENT & ARCHITECTURE (ZERO REWARD SHAPING) ---
R, DT, T_MAX, DELTA_CAP = 10.0, 0.08, 80.0, 0.03
K_SHARK = 3.60
N_POP, ELITE_FRAC, SEED = 150, 0.10, 42
NI, NH, NO = 4, 10, 2
NW = NI*NH + NH + NH*NO + NO

# Fixed Train (4 angles) & Held-Out Test (8 angles)
TRAIN_ANGLES = [-0.75*np.pi, -0.25*np.pi, 0.25*np.pi, 0.75*np.pi]
TEST_ANGLES  = [(-0.875 + 0.25*i)*np.pi for i in range(8)]

def wrap(a):
    return (a + np.pi) % (2 * np.pi) - np.pi

def forward(w, x, y, sh):
    r = np.hypot(x, y)
    th = np.arctan2(y, x) if r > 1e-6 else 0.0
    gp = wrap(th - sh)
    inp = np.array([r / R, np.cos(gp), np.sin(gp), 1.0])
    W1 = w[:NI*NH].reshape(NH, NI)
    b1 = w[NI*NH : NI*NH+NH]
    W2 = w[NI*NH+NH : NI*NH+NH+NH*NO].reshape(NO, NH)
    b2 = w[NI*NH+NH+NH*NO:]
    h = np.tanh(W1 @ inp + b1)
    o = W2 @ h + b2
    return th + np.arctan2(o[0], o[1])

def run_episode(w, sh_init, record=False):
    x, y, sh, t, rmax = 0.0, 0.0, sh_init, 0.0, 0.0
    traj = [(x, y)] if record else None
    while t < T_MAX:
        hd = forward(w, x, y, sh)
        dx, dy = np.cos(hd)*DT, np.sin(hd)*DT
        f, hit = 1.0, False
        if np.hypot(x + dx, y + dy) >= R:
            a = dx*dx + dy*dy
            b = 2*(x*dx + y*dy)
            c = x*x + y*y - R*R
            f = (-b + np.sqrt(max(0.0, b*b - 4*a*c))) / (2*a)
            hit = True
        x += f*dx; y += f*dy; t += f*DT
        if record: traj.append((x, y))
        r = np.hypot(x, y)
        th = np.arctan2(y, x)
        gp = wrap(th - sh)
        sh += np.sign(gp if gp != 0 else 1.0) * min(K_SHARK/R * DT * f, abs(gp))
        if hit:
            margin_m = (abs(wrap(th - sh)) - DELTA_CAP) * R
            return margin_m - 0.02*t, margin_m, margin_m > 0, traj, sh
        rmax = max(rmax, r)
    # Timeout penalty based strictly on unreached radial distance
    return -R*(1.0 - rmax/R) - 2.0, -R*(1.0 - rmax/R), False, traj, sh

# --- 2. REPRODUCIBLE GENETIC EVOLUTION ---
rng = np.random.default_rng(SEED)
pop = [rng.normal(0, 0.8, NW) for _ in range(N_POP)]
baseline_m = R * (np.pi / K_SHARK - DELTA_CAP)

for gen in range(1, 31):
    scored = []
    for w in pop:
        res = [run_episode(w, a) for a in TRAIN_ANGLES]
        scored.append((np.mean([r[0] for r in res]), np.mean([r[1] for r in res]), w))
    scored.sort(key=lambda x: x[0], reverse=True)
    champ = scored[0][2]
    test_res = [run_episode(champ, a) for a in TEST_ANGLES]
    test_margin = np.mean([r[1] for r in test_res])
    test_esc = np.mean([r[2] for r in test_res]) * 100
    if gen == 1 or gen % 5 == 0:
        print(f"Gen {gen:02d} | Held-Out Test Margin: {test_margin:+.2f}m | Escaped: {test_esc:.0f}% (Baseline: {baseline_m:+.2f}m)")

    n_el = max(4, int(N_POP * ELITE_FRAC))
    elites = [s[2] for s in scored[:n_el]]
    sigma = max(0.04, 0.35 * (0.965 ** gen))
    next_pop = [e.copy() for e in elites]
    while len(next_pop) < N_POP:
        p1, p2 = elites[rng.integers(n_el)], elites[rng.integers(n_el)]
        mask = rng.random(NW) < 0.5
        child = np.where(mask, p1, p2) + (rng.random(NW) < 0.32) * rng.normal(0, sigma, NW)
        next_pop.append(child)
    pop = next_pop
